20/**


21 * Step18 · 最小完整 Agent。


22 * <p>


23 * 一个 chat(input) 调用代表一个用户回合（turn），核心生命周期是：


24 *


25 * <pre>


26 * 输入守卫 → 组装上下文 → 标准工具协议循环 → 输出守卫 → 回写历史 → 按需压缩


27 * </pre>


28 *


29 * <p>模型不通过手写 JSON 表达“要调用哪个工具”，而是每次请求都收到


30 * {@code tools} Schema；模型返回结构化 {@code tool_calls}，宿主执行后用


31 * {@code role=tool + tool_call_id} 原样回填。这样保留了“谁请求调用、


32 * 结果属于哪次调用”的协议关系，与 OpenAI 兼容接口和现代 Agent Harness 一致。


33 *


34 * <p>状态被拆成三类：{@link #lastTrace} 是当前回合的可观测事件；


35 * {@code history} 是跨轮对话摘要级上下文；{@code memory} 是模型可通过


36 * 工具读写的结构化事实。它们解决的问题不同，不能互相替代。


37 *


38 * <p>教学简化：memory 是进程内 KV，压缩阈值按消息数估算，敏感词用正则。


39 * 生产系统应替换为持久化记忆、真实 token 计数和专业安全/审批组件。


40 */


41public interface FullAgent extends Agent {


42 


43    /** 单回合模型请求和工具执行的总步数上界，防止模型陷入无限工具调用。 */


44    int MAX_STEPS = 5;


45 


46    /** 模型前输入守卫阈值；超限输入直接拒绝，不消耗模型预算。 */


47    int MAX_INPUT_LEN = 300;


48 


49    /** 跨轮历史消息条数阈值；超过后把较旧轮次压缩成一段摘要。 */


50    int COMPACT_THRESHOLD = 10;


51 


52    /** 压缩时原样保留的最近消息数，保证最新任务状态不被摘要失真。 */


53    int KEEP_RECENT = 4;


54 


55    /**


56     * 当前回合内保留最近 N 条工具观察；更早的可重取结果清成占位符。


57     * 清理内容但不删除消息，避免破坏 assistant(tool_calls) 与 tool 的配对。


58     */


59    int KEEP_TOOL_RESULTS = 2;

61    class Impl implements FullAgent {


62 


63        /**


64         * 稳定系统提示词只约定行为和工具使用边界。


65         * 工具名、参数结构和必填项放在 toolSchemas() 中，


66         * 避免在提示词里维护第二份容易过期的工具清单。


67         */


68        private static final String SYSTEM_PROMPT = """


69                你是 ToyAgent 综合智能体。


70                小事闲聊时直接回答；需要外部状态、计算或持久记忆时调用工具。


71                用户说出可跨轮使用的稳定事实时，用 memory 工具保存。


72                回答用户事实前，先用 memory 读取，不要凭历史猜测。


73                工具观察只用于支持回答，不要向用户复述原始日志。


74                """;


75 


76        /**


77         * 压缩提示词决定“丢什么、留什么”。这里偏向保留任务状态和关键事实，


78         * 丢弃寒暄和可重新获取的细节；摘要是有损操作，必须可评测。


79         */


80        private static final String COMPACT_PROMPT = """


81                压缩以下 Agent 会话。保留：目标、决定、稳定事实、已完成步骤、


82                失败原因、当前状态和下一步。丢弃寒暄和可重新获取的细节。


83                不确定的信息标为 UNVERIFIED。直接输出简洁摘要。


84                """;


85 


86        /** 教学用输入/输出敏感词；生产环境应使用专业安全服务和工具审批策略。 */


87        private static final Pattern SENSITIVE = Pattern.compile(


88                "hack|攻击|爆破|破解|密码", Pattern.CASE_INSENSITIVE);


89 


90        /** 可替换的模型网关；Agent 只依赖标准对话和工具协议，不绑定具体厂商。 */


91        private final ChatModel model;


92 


93        /** 当前回合的执行轨迹，用于课堂演示、调试和生产审计。 */


94        private final List<Map<String, Object>> lastTrace = new ArrayList<>();


95 


96        /**


97         * 跨轮对话历史。为了控制上下文，这里只回写每轮最终输入/输出；


98         * 中间工具请求和结果保留在当前回合的 messages 中，并由 trace 记录。


99         */


100        private final List<Message> history = new ArrayList<>();


101 


102        /** 模型驱动的跨轮记忆后端；教学用 Map，生产可替换为文件或数据库，分布式可引入 redis。 */


103        private final Map<String, String> memory = new LinkedHashMap<>();

106            this.model = model;

111            return "Step18 · 最小完整 Agent";

114        @Override


115        public String chat(String input) {


116            // 阶段 1：模型前守卫。低成本规则先执行，被拦截输入不进入 Agent Loop。


117            lastTrace.clear();


118            if (input == null || input.isBlank()) {


119                return "请输入内容。";


120            }


121            if (input.length() > MAX_INPUT_LEN) {


122                trace("guard", "输入守卫", "长度 " + input.length() + " 超过 " + MAX_INPUT_LEN + "，未调用模型。");


123                return "输入过长，请精简后重试。";


124            }


125            Matcher unsafe = SENSITIVE.matcher(input);


126            if (unsafe.find()) {


127                trace("guard", "输入守卫", "命中敏感词「" + unsafe.group() + "」，未调用模型。");


128                return "该请求涉及不安全操作，已被输入守卫拦截。";


129            }


130            trace("guard", "输入守卫", "长度和安全检查通过。");


131 


132            // 阶段 2：组装当前回合上下文。系统提示词在前，历史居中，新输入最后。


133            List<Message> messages = new ArrayList<>();


134            messages.add(Message.system(SYSTEM_PROMPT));


135            messages.addAll(history);


136            messages.add(Message.user(input));


137 


138            // 阶段 3：循环推理与工具执行，得到候选答案。


139            String answer = runAgentLoop(messages);


140 


141            // 阶段 4：输出仍不可信，先守卫；再只把最终问答写入跨轮历史。


142            answer = outputGuard(answer);


143            rememberTurn(input, answer);


144 


145            // 阶段 5：本轮结束后治理历史，避免下一轮上下文无限增长。


146            compactHistoryIfNeeded();


147            return answer;


148        }

150        private String runAgentLoop(List<Message> messages) {


151            /*


152             * 标准 Agent Loop：


153             * 1. 带 tools 请求模型；


154             * 2. toolCalls 为空，说明模型已给出最终答案；


155             * 3. 不为空时，先原样保存 assistant(tool_calls)，再逐个执行工具；


156             * 4. 每个观察都作为 role=tool 消息、按 tool_call_id 回填；


157             * 5. 继续循环，直到模型收敛或触发 MAX_STEPS。


158             *


159             * 关键不变量：不能把工具结果改写成普通 user 文本，


160             * 也不能删除 assistant(tool_calls) 而单独保留 tool 消息。


161             */


162            for (int step = 1; step <= MAX_STEPS; step++) {


163                clearOldToolResults(messages);


164                ChatResponse response = model.chat(messages, toolSchemas());


165                trace("model", "Model " + step,


166                        response.content() == null || response.content().isBlank()


167                                ? "返回 " + response.toolCalls().size() + " 个结构化 tool_calls"


168                                : "返回最终文本");


169 


170                if (response.toolCalls().isEmpty()) {


171                    trace("final", "Final Answer", response.content());


172                    return response.content() == null ? "" : response.content();


173                }


174 


175                messages.add(new Message("assistant", response.content(), response.toolCalls(), null));


176                for (ToolCall call : response.toolCalls()) {


177                    Map<String, Object> args = parseArguments(call.argumentsJson());


178                    trace("tool_call", "ToolCall " + step, call.name() + " " + args);


179                    String observation = executeTool(call.name(), args);


180                    trace("tool_result", "ToolResult " + step, observation);


181                    messages.add(Message.tool(call.id(), observation));


182                }


183            }


184            String fused = "已达最大步数 " + MAX_STEPS + "，循环保险丝熔断。";


185            trace("guard", "保险丝", fused);


186            return fused;


187        }

189        /**


190         * 宿主工具边界：模型只能给出名字和参数，真实执行权始终在这里。


191         * 异常也作为 observation 返回，让模型有机会修正参数或换用工具。


192         */


193        private String executeTool(String name, Map<String, Object> args) {


194            try {


195                return switch (name == null ? "" : name) {


196                    case "get_weather" -> ReActAgent.MockWeather.query(text(args, "city"));


197                    case "calculator" -> String.valueOf(calc(text(args, "expression")));


198                    case "get_time" -> LocalDateTime.now()


199                            .format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"));


200                    case "memory" -> executeMemory(args);


201                    default -> "未知工具: " + name;


202                };


203            } catch (Exception exception) {


204                return "工具执行失败: " + exception.getMessage();


205            }


206        }


207 


208        /**


209         * memory 是模型显式调用的工具，不是启动时无条件注入的全部历史。


210         * 模型决定何时写、读什么；宿主决定存储位置和访问规则。


211         */


212        private String executeMemory(Map<String, Object> args) {


213            String command = text(args, "command");


214            String key = text(args, "key");


215            return switch (command) {


216                case "write" -> {


217                    String value = text(args, "value");


218                    if (key.isBlank() || value.isBlank()) {


219                        yield "工具执行失败: key/value 不能为空";


220                    }


221                    memory.put(key, value);


222                    yield "已保存 " + key + "（当前记忆 " + memory.size() + " 项）。";


223                }


224                case "read" -> memory.containsKey(key)


225                        ? key + "=" + memory.get(key)


226                        : key + " 不存在。当前记忆键: " + memory.keySet();


227                case "keys" -> memory.isEmpty() ? "记忆为空。" : "记忆键: " + memory.keySet();


228                case "delete" -> memory.remove(key) == null


229                        ? key + " 不存在。"


230                        : "已删除 " + key + "。";


231                default -> "未知记忆命令: " + command;


232            };


233        }

235        /** 解析模型返回的 argumentsJson；协议约定它必须是 JSON 对象。 */


236        private Map<String, Object> parseArguments(String argumentsJson) {


237            if (argumentsJson == null || argumentsJson.isBlank()) {


238                return Map.of();


239            }


240            Object parsed = Json.parse(argumentsJson);


241            return parsed instanceof Map<?, ?> map ? asObjectMap(map) : Map.of();


242        }


243 


244        /** 把不可变的解析结果转换为工具执行器可直接访问的 Map。 */


245        private Map<String, Object> asObjectMap(Map<?, ?> source) {


246            Map<String, Object> result = new LinkedHashMap<>();


247            source.forEach((key, value) -> result.put(String.valueOf(key), value));


248            return result;


249        }

251        /** 统一把工具参数转换为字符串；缺失参数按空串处理并由工具校验。 */


252        private String text(Map<String, Object> args, String key) {


253            Object value = args.get(key);


254            return value == null ? "" : String.valueOf(value);


255        }

257        /** 模型输出同样不可信；进入用户界面和跨轮历史前再做一次检查。 */


258        private String outputGuard(String answer) {


259            if (answer != null && SENSITIVE.matcher(answer).find()) {


260                trace("guard", "输出守卫", "答案命中敏感词，已替换。");


261                return "该内容不宜展示，已由输出守卫处理。";


262            }


263            return answer == null ? "" : answer;


264        }

266        /** 回合收口：只保存最终问答，避免中间工具载荷污染后续轮次。 */


267        private void rememberTurn(String input, String answer) {


268            history.add(Message.user(input));


269            history.add(Message.assistant(answer));


270            trace("memory", "会话记忆", "本轮写入历史，当前 " + history.size() + " 条。");


271        }

273        /**


274         * 历史压缩（compaction）：较旧轮次交给模型生成摘要，最近轮次原样保留。


275         * memory 中的结构化事实不参与压缩，因此关键事实仍可跨轮读取。


276         * 摘要模型失败时退化为滑动窗口，这是确定性更高但有硬遗忘的备选策略。


277         */


278        private void compactHistoryIfNeeded() {


279            if (history.size() <= COMPACT_THRESHOLD) {


280                return;


281            }


282            int split = history.size() - KEEP_RECENT;


283            List<Message> older = new ArrayList<>(history.subList(0, split));


284            List<Message> recent = new ArrayList<>(history.subList(split, history.size()));


285            try {


286                String summary = model.chat(List.of(


287                        Message.system(COMPACT_PROMPT),


288                        Message.user(renderTranscript(older))


289                ));


290                history.clear();


291                history.add(Message.user("[对话摘要]\n" + summary));


292                history.addAll(recent);


293                trace("compact", "上下文压缩",


294                        older.size() + " 条历史压缩为摘要，保留最近 " + recent.size() + " 条。");


295            } catch (Exception exception) {


296                history.clear();


297                history.addAll(recent);


298                trace("compact", "上下文压缩",


299                        "摘要调用失败，退化为滑动窗口；保留最近 " + recent.size() + " 条。");


300            }


301        }

303        /** 把消息列表渲染为压缩器可读的文本转写。 */


304        private String renderTranscript(List<Message> messages) {


305            StringBuilder transcript = new StringBuilder();


306            for (Message message : messages) {


307                transcript.append('[').append(message.role()).append("] ")


308                        .append(message.content() == null ? "" : message.content())


309                        .append('\n');


310            }


311            return transcript.toString();


312        }

314        /**


315         * 工具结果清理（tool clearing）：只清可重取的大块观察，不删除协议消息。


316         * 每次调模型前从后往前统计 role=tool；超过最近 N 条的旧结果替换为占位符。


317         * 这样既减小上下文，又保持 assistant(tool_calls) 与 tool 结果的合法配对。


318         *


319         * <p>举例：KEEP_TOOL_RESULTS=2，当前回合已有 3 次工具调用：


320         *


321         * <pre>


322         * 清理前：


323         * assistant(tool_calls=[call-1: get_weather("北京")])


324         * tool(call-1, "北京晴，26℃")                         ← 第 3 新，清理


325         * assistant(tool_calls=[call-2: calculator("12*8")])


326         * tool(call-2, "96.0")                                ← 第 2 新，保留


327         * assistant(tool_calls=[call-3: get_time()])


328         * tool(call-3, "2026-11-20 10:00:00")                 ← 第 1 新，保留


329         *


330         * 清理后：


331         * assistant(tool_calls=[call-1: get_weather("北京")])  ← 保留请求记录


332         * tool(call-1, "[工具结果已清理，需要时重新调用]")      ← 只替换结果内容


333         * assistant(tool_calls=[call-2: calculator("12*8")])


334         * tool(call-2, "96.0")


335         * assistant(tool_calls=[call-3: get_time()])


336         * tool(call-3, "2026-11-20 10:00:00")


337         * </pre>


338         *


339         * <p>因此“清理”不是删除工具调用历史，而是把旧观察压缩成占位符；


340         * assistant 的调用请求、tool 消息角色和 tool_call_id 配对关系都保留。


341         */


342        private void clearOldToolResults(List<Message> messages) {


343            int count = 0;


344            int cleared = 0;


345            for (int index = messages.size() - 1; index >= 0; index--) {


346                Message message = messages.get(index);


347                if (!"tool".equals(message.role())) {


348                    continue;


349                }


350                count++;


351                if (count > KEEP_TOOL_RESULTS) {


352                    messages.set(index, Message.tool(message.toolCallId(), "[工具结果已清理，需要时重新调用]"));


353                    cleared++;


354                }


355            }


356            if (cleared > 0) {


357                trace("context", "工具结果清理", "清理 " + cleared + " 条旧观察，保留最近 "


358                        + KEEP_TOOL_RESULTS + " 条，协议配对不变。");


359            }


360        }

362        /** 教学用单步四则运算；不使用反射或脚本引擎，避免代码执行面。 */


363        private double calc(String expression) {


364            Matcher matcher = Pattern.compile("\\s*(\\d+(?:\\.\\d+)?)\\s*([+\\-*/])\\s*(\\d+(?:\\.\\d+)?)\\s*")


365                    .matcher(expression);


366            if (!matcher.matches()) {


367                throw new IllegalArgumentException("无法解析表达式: " + expression);


368            }


369            double left = Double.parseDouble(matcher.group(1));


370            double right = Double.parseDouble(matcher.group(3));


371            return switch (matcher.group(2)) {


372                case "+" -> left + right;


373                case "-" -> left - right;


374                case "*" -> left * right;


375                case "/" -> {


376                    if (right == 0) {


377                        throw new ArithmeticException("除数不能为 0");


378                    }


379                    yield left / right;


380                }


381                default -> throw new IllegalArgumentException("不支持的运算符");


382            };


383        }

385        /**


386         * 工具清单是模型的唯一“能力视图”。每个工具用 JSON Schema 声明


387         * 名称、用途、参数和必填项；宿主 switch 只是这些 Schema 的执行后端。


388         */


389        private List<Map<String, Object>> toolSchemas() {


390            return List.of(


391                    function("get_weather", "查询城市天气。", stringProperty("city", "城市名"), "city"),


392                    function("calculator", "计算一次四则运算。", stringProperty("expression", "表达式，如 12*8"), "expression"),


393                    function("get_time", "获取当前本地时间。", Map.of(), null),


394                    function("memory", "跨轮持久记忆的读写接口。",


395                            Map.of(


396                                    "command", Map.of("type", "string",


397                                            "enum", List.of("write", "read", "keys", "delete"),


398                                            "description", "记忆操作"),


399                                    "key", Map.of("type", "string", "description", "记忆键"),


400                                    "value", Map.of("type", "string", "description", "write 时的值")),


401                            "command"));


402        }

404        private Map<String, Object> function(String name, String description,


405                                             Map<String, Object> properties, String... required) {


406            /** 构造 OpenAI tools 协议里的一个 function 定义。 */


407            Map<String, Object> parameters = new LinkedHashMap<>();


408            parameters.put("type", "object");


409            parameters.put("properties", properties);


410            if (required != null && required.length > 0) {


411                parameters.put("required", List.of(required));


412            }


413            Map<String, Object> function = new LinkedHashMap<>();


414            function.put("name", name);


415            function.put("description", description);


416            function.put("parameters", parameters);


417            return Map.of("type", "function", "function", function);


418        }

420        private Map<String, Object> stringProperty(String name, String description) {


421            /** 构造一个简单字符串参数 Schema。 */


422            return Map.of(name, Map.of("type", "string", "description", description));
